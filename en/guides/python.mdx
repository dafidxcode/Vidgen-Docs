```python vidgen.py theme={null}
import os
import time
from typing import Any

import requests

BASE = "https://vidgen.id/api/v1"


class VidgenError(RuntimeError):
    def __init__(self, status: int, body: dict[str, Any] | None):
        self.status = status
        self.body = body or {}
        super().__init__(self.body.get("error") or f"HTTP {status}")


class Vidgen:
    def __init__(self, api_key: str | None = None):
        self.session = requests.Session()
        self.session.headers["Authorization"] = f"Bearer {api_key or os.environ['VIDGEN_API_KEY']}"

    def _request(self, method: str, path: str, **kwargs: Any) -> dict[str, Any]:
        res = self.session.request(method, f"{BASE}{path}", timeout=kwargs.pop("timeout", 120), **kwargs)
        try:
            payload = res.json()
        except ValueError:
            payload = None
        if not res.ok:
            raise VidgenError(res.status_code, payload)
        return payload

    def generate(self, model: str, prompt: str, **kwargs: Any) -> dict[str, Any]:
        body = {"model": model, "prompt": prompt, **kwargs}
        return self._request("POST", "/generate", json=body)["job"]

    def job(self, job_id: str) -> dict[str, Any]:
        return self._request("GET", f"/jobs/{job_id}")["job"]

    def upload(self, path: str, model: str | None = None) -> dict[str, Any]:
        files = {"file": open(path, "rb")}
        data = {"model": model} if model else None
        try:
            return self._request("POST", "/upload", files=files, data=data)["asset"]
        finally:
            files["file"].close()

    def credits(self) -> dict[str, Any]:
        return self._request("GET", "/user/credits")

    def models(self) -> dict[str, Any]:
        return self._request("GET", "/models")["models"]

    def wait_for_job(
        self,
        job_id: str,
        interval: float = 5.0,
        cap: float = 30.0,
        timeout: float = 2 * 60 * 60,
    ) -> dict[str, Any]:
        deadline = time.monotonic() + timeout
        delay = interval
        while True:
            job = self.job(job_id)
            if job["status"] in {"succeeded", "failed", "cancelled"}:
                return job
            if time.monotonic() + delay > deadline:
                return job
            time.sleep(delay)
            delay = min(delay * 1.6, cap)


def estimate_credits(model: dict[str, Any], parameters: dict[str, str] | None = None) -> float:
    """Mirror the server's price lookup before the job is sent.

    quality:resolution -> resolution:duration (both orders) -> duration ->
    resolution -> creditsDefault. `model` is one entry from api.models().
    Models flagged `unlimitedOnly` are not handled here: the charge is 0 when
    the plan is active, and the request is rejected 403 when it is not.
    """
    parameters = parameters or {}
    matrix = model.get("priceMatrix") or {}
    res = parameters.get("resolution")
    res_low = res.lower() if res else None
    dur = parameters.get("duration")
    dur_keys = [dur.rstrip("sS"), dur.rstrip("sS") + "s"] if dur else []
    quality = (parameters.get("quality") or "").lower()

    def num(value: Any) -> float | None:
        return float(value) if isinstance(value, (int, float)) else None

    if quality and res:
        for key in (f"{quality}:{res}", f"{quality}:{res_low}"):
            hit = num(matrix.get(key))
            if hit is not None:
                return hit
    if res_low and dur_keys:
        for key in dur_keys:
            for combo in (f"{res_low}:{key}", f"{key}:{res_low}",
                          f"{res_low}:{key.lower()}", f"{key.lower()}:{res_low}"):
                hit = num(matrix.get(combo))
                if hit is not None:
                    return hit
    for key in dur_keys:
        hit = num(matrix.get(key))
        if hit is None:
            hit = num(matrix.get(key.lower()))
        if hit is not None:
            return hit
    if res:
        hit = num(matrix.get(res))
        if hit is None and res_low:
            hit = num(matrix.get(res_low))
        if hit is not None:
            return hit
    return float(model["creditsDefault"])
```

## Usage

```python theme={null}
import uuid

import requests

from vidgen import Vidgen, VidgenError, estimate_credits

api = Vidgen()

MODEL = "seedance-2.5-promo"
PARAMS = {"duration": "5", "ratio": "16:9", "resolution": "480p"}

model = next((m for m in api.models()["video"] if m["id"] == MODEL), None)
if model is None:
    raise SystemExit(f"{MODEL} is not available right now")
if model["unlimitedOnly"]:
    raise SystemExit("this model needs an active Unlimited plan")

print("estimated credits:", estimate_credits(model, PARAMS))  # 9.0

asset = api.upload("character.png", model=MODEL)

job = api.generate(
    MODEL,
    prompt="this character waves at the camera, soft studio background",
    references=[asset],
    parameters=PARAMS,
    idempotencyKey=f"batch-7-{uuid.uuid4().hex[:12]}",
)

finished = api.wait_for_job(job["id"])

if finished["status"] == "succeeded":
    with open(f"result-{finished['id']}.mp4", "wb") as f:
        f.write(requests.get(finished["result"]["url"], timeout=300).content)
else:
    print(finished["errorMessage"])
```

<Warning>
  The field name in the JSON body is `idempotencyKey` (camelCase), not `idempotency_key`. A typo means the server generates its own key and your retry is unprotected.
</Warning>

## Batch pattern

Do not guess the duration. Take the first value the model actually supports from the catalog, then compute the allowance from **one** `credits()` call:

```python theme={null}
import time

from vidgen import Vidgen, VidgenError, estimate_credits


def generate_batch(api: Vidgen, prompts: list[str], model_id: str) -> list[dict]:
    model = next((m for m in api.models()["video"] if m["id"] == model_id), None)
    if model is None:
        raise ValueError(f"{model_id} is not available right now")
    if model["unlimitedOnly"]:
        raise ValueError("Unlimited models are not suitable for a paid batch")

    durations = model["capabilities"]["durations"]
    resolutions = model["capabilities"]["resolutions"]
    params: dict[str, str] = {}
    if durations:
        params["duration"] = durations[0]
    if resolutions:
        params["resolution"] = resolutions[0]

    price = estimate_credits(model, params)
    account = api.credits()
    limits = account["limits"]
    balance = float(account["user"]["credits"])

    slot_allowance = max(0, limits["maxConcurrentJobs"] - limits["currentActiveJobs"])
    allowance = min(
        limits["dailyVideosRemaining"],
        slot_allowance,
        int(balance // price) if price else len(prompts),
    )

    jobs: list[dict] = []
    for prompt in prompts[:allowance]:
        try:
            jobs.append(api.generate(model_id, prompt=prompt, parameters=params))
        except VidgenError as err:
            if err.status in (402, 403, 409):
                break
            if err.status == 429:
                time.sleep(30)
                continue
            raise
    return jobs
```

<Note>
  `maxConcurrentJobs` is only enforced for accounts with an active Unlimited quota; on pay-per-credit accounts the value is reported but does not block. Limiting yourself with that number is still polite and keeps the shared queue short.
</Note>

<Tip>
  For async services (FastAPI, Django), call `wait_for_job` in a separate worker, not in the request handler — two hours of polling would exhaust your connections.
</Tip>
