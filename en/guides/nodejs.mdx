The complete client below uses only the built-in `fetch` — no extra packages, no build step. Save it as `vidgen.js` and run it with Node 18 or newer.

```js vidgen.js theme={null}
const BASE = "https://vidgen.id/api/v1";

export class VidgenError extends Error {
  constructor(status, body) {
    super(body?.error ?? `HTTP ${status}`);
    this.status = status;
    this.body = body ?? {};
  }
}

async function call(path, { method = "GET", body, form } = {}) {
  const headers = { Authorization: `Bearer ${process.env.VIDGEN_API_KEY}` };
  if (body) headers["Content-Type"] = "application/json";

  const res = await fetch(`${BASE}${path}`, {
    method,
    headers,
    body: form ?? (body ? JSON.stringify(body) : undefined),
  });

  const payload = await res.json().catch(() => null);
  if (!res.ok) throw new VidgenError(res.status, payload);
  return payload;
}

export const vidgen = {
  /** Create a job. Returns a job object with status `queued`. */
  generate: (input) => call("/generate", { method: "POST", body: input }).then((r) => r.job),

  /** Fetch the latest status; the server also settles `processing` jobs. */
  job: (id) => call(`/jobs/${id}`).then((r) => r.job),

  /** Upload a reference from a Blob/Buffer. Include `model` so the duration rules are right. */
  upload: (file, model, filename = "reference.bin") => {
    const form = new FormData();
    form.append("file", file, filename);
    if (model) form.append("model", model);
    return call("/upload", { method: "POST", form }).then((r) => r.asset);
  },

  /** Balance, Unlimited status, and the quota left for today. */
  credits: () => call("/user/credits"),

  /** Active model catalog — public, still send a key for consistency. */
  models: () => call("/models").then((r) => r.models),
};

/** Poll with backoff until the job reaches a final status. */
export async function waitForJob(id, { interval = 5_000, cap = 30_000, timeout = 2 * 60 * 60_000 } = {}) {
  const deadline = Date.now() + timeout;
  let delay = interval;

  for (;;) {
    const job = await vidgen.job(id);
    if (job.status === "succeeded" || job.status === "failed" || job.status === "cancelled") return job;
    if (Date.now() + delay > deadline) return job;
    await new Promise((r) => setTimeout(r, delay));
    delay = Math.min(delay * 1.6, cap);
  }
}

/**
 * Credit estimate before the job is sent. Mirrors the server's price lookup:
 * quality:resolution -> resolution:duration (both orders) -> duration ->
 * resolution -> creditsDefault. `model` is one entry from GET /models.
 */
export function estimateCredits(model, parameters = {}) {
  const matrix = model.priceMatrix ?? {};
  const res = parameters.resolution;
  const resLow = res?.toLowerCase();
  const dur = parameters.duration ? String(parameters.duration).replace(/s$/i, "") : undefined;
  const durKeys = dur ? [dur, `${dur}s`] : [];
  const quality = parameters.quality?.toLowerCase();
  const num = (v) => (typeof v === "number" ? v : undefined);

  if (quality && res) {
    const hit = num(matrix[`${quality}:${res}`]) ?? num(matrix[`${quality}:${resLow}`]);
    if (hit !== undefined) return hit;
  }
  if (resLow && dur) {
    for (const k of durKeys) {
      const hit =
        num(matrix[`${resLow}:${k}`]) ?? num(matrix[`${k}:${resLow}`]) ??
        num(matrix[`${resLow}:${k.toLowerCase()}`]) ?? num(matrix[`${k.toLowerCase()}:${resLow}`]);
      if (hit !== undefined) return hit;
    }
  }
  for (const k of durKeys) {
    const hit = num(matrix[k]) ?? num(matrix[k.toLowerCase()]);
    if (hit !== undefined) return hit;
  }
  if (res) {
    const hit = num(matrix[res]) ?? num(matrix[resLow]);
    if (hit !== undefined) return hit;
  }
  return model.creditsDefault;
}
```

<Note>
  `estimateCredits` ignores `unlimitedOnly`. For a flagged model, the real charge is `0` when the Unlimited plan is active and the request is rejected `403` when it is not — check that flag before using this number.
</Note>

## Usage

```js theme={null}
import { vidgen, waitForJob, estimateCredits } from "./vidgen.js";
import { readFile, writeFile } from "node:fs/promises";
import { randomUUID } from "node:crypto";

const MODEL = "seedance-2.5-promo";
const PARAMS = { duration: "5", ratio: "16:9", resolution: "480p" };

// Public catalog: no key needed, but the client above still sends one.
const catalog = await vidgen.models();
const model = catalog.video.find((m) => m.id === MODEL);
if (!model) throw new Error(`${MODEL} is not available right now`);
if (model.unlimitedOnly) throw new Error("this model needs an active Unlimited plan");

console.log("estimated credits:", estimateCredits(model, PARAMS)); // 9

const bytes = await readFile("character.png");
const character = await vidgen.upload(new Blob([bytes], { type: "image/png" }), MODEL, "character.png");

const job = await vidgen.generate({
  model: MODEL,
  prompt: "this character waves at the camera, soft studio background",
  references: [character],
  parameters: PARAMS,
  idempotencyKey: `client-42-${randomUUID().slice(0, 8)}`,
});

const finished = await waitForJob(job.id);

if (finished.status === "succeeded") {
  const result = await fetch(finished.result.url).then((r) => r.arrayBuffer());
  await writeFile(`result-${finished.id}.mp4`, result);
} else {
  console.error(finished.errorMessage, "credits refunded:", finished.creditsCharged === 0);
}
```

<Warning>
  Node's `fetch()` does not support `file://` URLs. Read the file with `node:fs/promises` and wrap it in a `Blob` as shown. `crypto.randomUUID()` only became global in Node 19 — import it from `node:crypto` so it works on Node 18.
</Warning>

## Handling errors

```js theme={null}
try {
  await vidgen.generate({ model: "seedance-2.5-promo", prompt: "..." });
} catch (err) {
  switch (err.status) {
    case 401: /* wrong or revoked key — do not retry */ throw err;
    case 402: /* balance exhausted: err.body.requiredCredits, err.body.currentCredits */ break;
    case 403: /* err.body.code === "unlimited_required": quota model, not for this account */ break;
    case 409: /* inactive model — refetch the catalog */ break;
    case 429: /* capacity: wait, then retry with the same idempotencyKey */ break;
    case 503: /* maintenance: defer the job to the next window */ break;
  }
}
```

<Warning>
  The diagnostic fields (`requiredCredits`, `maxConcurrent`, `dailyVideosUsed`) are on `err.body`, not in the message. Do not parse the error string for logic.
</Warning>

## Type Safety

A short set of response types that is enough for most integrations:

```ts types.ts theme={null}
export type JobStatus = "queued" | "dispatching" | "processing" | "succeeded" | "failed" | "cancelled";

export interface Job {
  id: string;
  status: JobStatus;
  model: string;
  category: "Video" | "Image" | "Audio";
  prompt: string;
  queuePosition?: number | null;
  result: { url: string; duration?: number; width?: number; height?: number } | null;
  errorMessage: string | null;
  creditsCharged: number;
  createdAt: string;
  updatedAt?: string;
}

export interface Asset {
  id: string;
  url: string;
  file_url: string;
  type: "image" | "video" | "audio";
  filename: string;
  duration?: number | null;
  width?: number | null;
  height?: number | null;
}

export interface CatalogModel {
  id: string;
  label: string;
  category: "Video" | "Image" | "Audio";
  mode: "video" | "image" | "audio";
  creditsDefault: number;
  unlimitedOnly: boolean;
  priceMatrix: Record<string, number>;
  capabilities: {
    durations: string[];
    ratios: string[];
    resolutions: string[];
    maxReferenceImages: number;
    maxReferenceVideos: number;
    maxReferenceAudios: number;
    minReferenceVideos?: number;
    qualities?: string[];
    backgrounds?: string[];
    outputFormats?: string[];
    supportsNegativePrompt?: boolean;
    supportsSeed?: boolean;
    feature?: "create" | "motion";
    motion?: { minDurationSec?: number; maxDurationSec?: number; durationFromReference?: boolean };
    modes?: string[];
    versions?: string[];
  };
}
```

<Note>
  The client above calls Vidgen from the server. From a browser the request is blocked by CORS — keep the key and these calls on your backend.
</Note>
