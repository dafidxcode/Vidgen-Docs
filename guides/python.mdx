```python vidgen.py theme={null}
import os
import time
from typing import Any

import requests

BASE = "https://vidgen.id/api/v1"


class VidgenError(RuntimeError):
    def __init__(self, status: int, body: dict[str, Any] | None):
        self.status = status
        self.body = body or {}
        super().__init__(self.body.get("error") or f"HTTP {status}")


class Vidgen:
    def __init__(self, api_key: str | None = None):
        self.session = requests.Session()
        self.session.headers["Authorization"] = f"Bearer {api_key or os.environ['VIDGEN_API_KEY']}"

    def _request(self, method: str, path: str, **kwargs: Any) -> dict[str, Any]:
        res = self.session.request(method, f"{BASE}{path}", timeout=kwargs.pop("timeout", 120), **kwargs)
        try:
            payload = res.json()
        except ValueError:
            payload = None
        if not res.ok:
            raise VidgenError(res.status_code, payload)
        return payload

    def generate(self, model: str, prompt: str, **kwargs: Any) -> dict[str, Any]:
        body = {"model": model, "prompt": prompt, **kwargs}
        return self._request("POST", "/generate", json=body)["job"]

    def job(self, job_id: str) -> dict[str, Any]:
        return self._request("GET", f"/jobs/{job_id}")["job"]

    def upload(self, path: str, model: str | None = None) -> dict[str, Any]:
        files = {"file": open(path, "rb")}
        data = {"model": model} if model else None
        try:
            return self._request("POST", "/upload", files=files, data=data)["asset"]
        finally:
            files["file"].close()

    def credits(self) -> dict[str, Any]:
        return self._request("GET", "/user/credits")

    def models(self) -> dict[str, Any]:
        return self._request("GET", "/models")["models"]

    def wait_for_job(
        self,
        job_id: str,
        interval: float = 5.0,
        cap: float = 30.0,
        timeout: float = 2 * 60 * 60,
    ) -> dict[str, Any]:
        deadline = time.monotonic() + timeout
        delay = interval
        while True:
            job = self.job(job_id)
            if job["status"] in {"succeeded", "failed", "cancelled"}:
                return job
            if time.monotonic() + delay > deadline:
                return job
            time.sleep(delay)
            delay = min(delay * 1.6, cap)


def estimate_credits(model: dict[str, Any], parameters: dict[str, str] | None = None) -> float:
    """Meniru urutan pencarian harga server sebelum job dikirim.

    quality:resolution -> resolution:duration (dua urutan) -> duration ->
    resolution -> creditsDefault. `model` adalah satu entri dari api.models().
    Model berflag `unlimitedOnly` tidak ditangani di sini: tagihannya 0 bila
    paket aktif, dan request ditolak 403 bila tidak.
    """
    parameters = parameters or {}
    matrix = model.get("priceMatrix") or {}
    res = parameters.get("resolution")
    res_low = res.lower() if res else None
    dur = parameters.get("duration")
    dur_keys = [dur.rstrip("sS"), dur.rstrip("sS") + "s"] if dur else []
    quality = (parameters.get("quality") or "").lower()

    def num(value: Any) -> float | None:
        return float(value) if isinstance(value, (int, float)) else None

    if quality and res:
        for key in (f"{quality}:{res}", f"{quality}:{res_low}"):
            hit = num(matrix.get(key))
            if hit is not None:
                return hit
    if res_low and dur_keys:
        for key in dur_keys:
            for combo in (f"{res_low}:{key}", f"{key}:{res_low}",
                          f"{res_low}:{key.lower()}", f"{key.lower()}:{res_low}"):
                hit = num(matrix.get(combo))
                if hit is not None:
                    return hit
    for key in dur_keys:
        hit = num(matrix.get(key))
        if hit is None:
            hit = num(matrix.get(key.lower()))
        if hit is not None:
            return hit
    if res:
        hit = num(matrix.get(res))
        if hit is None and res_low:
            hit = num(matrix.get(res_low))
        if hit is not None:
            return hit
    return float(model["creditsDefault"])
```

## Pemakaian

```python theme={null}
import uuid

import requests

from vidgen import Vidgen, VidgenError, estimate_credits

api = Vidgen()

MODEL = "seedance-2.5-promo"
PARAMS = {"duration": "5", "ratio": "16:9", "resolution": "480p"}

model = next((m for m in api.models()["video"] if m["id"] == MODEL), None)
if model is None:
    raise SystemExit(f"{MODEL} sedang tidak tersedia")
if model["unlimitedOnly"]:
    raise SystemExit("model ini butuh paket Unlimited aktif")

print("perkiraan kredit:", estimate_credits(model, PARAMS))  # 9.0

asset = api.upload("karakter.png", model=MODEL)

job = api.generate(
    MODEL,
    prompt="karakter ini melambaikan tangan ke kamera, latar studio lembut",
    references=[asset],
    parameters=PARAMS,
    idempotencyKey=f"batch-7-{uuid.uuid4().hex[:12]}",
)

hasil = api.wait_for_job(job["id"])

if hasil["status"] == "succeeded":
    with open(f"hasil-{hasil['id']}.mp4", "wb") as f:
        f.write(requests.get(hasil["result"]["url"], timeout=300).content)
else:
    print(hasil["errorMessage"])
```

<Warning>
  Nama field di body JSON adalah `idempotencyKey` (huruf besar di tengah), bukan `idempotency_key`. Salah tulis berarti server membuat key sendiri dan retry kamu tidak terlindungi.
</Warning>

## Pola batch

Jangan menebak durasi. Ambil nilai pertama yang benar-benar didukung model dari katalog, lalu hitung jatah dari **satu** panggilan `credits()`:

```python theme={null}
import time

from vidgen import Vidgen, VidgenError, estimate_credits


def generate_batch(api: Vidgen, prompts: list[str], model_id: str) -> list[dict]:
    model = next((m for m in api.models()["video"] if m["id"] == model_id), None)
    if model is None:
        raise ValueError(f"{model_id} sedang tidak tersedia")
    if model["unlimitedOnly"]:
        raise ValueError("model Unlimited tidak cocok untuk batch berbayar")

    durations = model["capabilities"]["durations"]
    resolutions = model["capabilities"]["resolutions"]
    params: dict[str, str] = {}
    if durations:
        params["duration"] = durations[0]
    if resolutions:
        params["resolution"] = resolutions[0]

    harga = estimate_credits(model, params)
    akun = api.credits()
    limits = akun["limits"]
    saldo = float(akun["user"]["credits"])

    jatah_slot = max(0, limits["maxConcurrentJobs"] - limits["currentActiveJobs"])
    jatah = min(limits["dailyVideosRemaining"], jatah_slot, int(saldo // harga) if harga else len(prompts))

    jobs: list[dict] = []
    for prompt in prompts[:jatah]:
        try:
            jobs.append(api.generate(model_id, prompt=prompt, parameters=params))
        except VidgenError as err:
            if err.status in (402, 403, 409):
                break
            if err.status == 429:
                time.sleep(30)
                continue
            raise
    return jobs
```

<Note>
  `maxConcurrentJobs` hanya ditegakkan untuk akun dengan kuota Unlimited aktif; pada akun berbayar per kredit nilainya dilaporkan tetapi tidak memblokir. Membatasi diri sendiri dengan angka itu tetap sopan dan menjaga antrean bersama tetap pendek.
</Note>

<Tip>
  Untuk layanan async (FastAPI, Django), panggil `wait_for_job` di worker terpisah, bukan di request handler — dua jam polling akan menghabiskan koneksi.
</Tip>
